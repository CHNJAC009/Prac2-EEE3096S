\section{SPI Waveform}
\label{sec:spi-waveform}

\paragraph{Group test byte.}
With $n_1 = \nOne$ and $n_2 = \nTwo$ (last three digits of each student number):
\begin{align}
  n_1 \oplus n_2 &= \text{\nOneHex} \oplus \text{\nTwoHex} = \text{\XorHex} \\
  B &= \left[(n_1 \oplus n_2) + \text{0x3D}\right] \bmod 256
     = \text{\SumHex} \bmod 256 = \text{\BRawHex}
\end{align}
\fillin{State whether the 0x00/0xFF rule applied.} The group test byte is
therefore $B = \text{\BHex} = \text{\BBin}_2$.

\paragraph{Capture.}
$B$ was written to \reg{SPI\_DR} every 100~ms with CS (PB12) held high so the
EEPROM ignored the transfer. CH1 was connected to SCK (PB13) and CH2 to MOSI
(PB15).

\paragraph{Frequency error.}
\begin{equation}
  \%\,\mathrm{error}
  = \frac{\lvert f_\mathrm{measured} - f_\mathrm{predicted}\rvert}{f_\mathrm{predicted}} \times 100
  = \frac{\lvert \text{\fSCKMeas} - \text{\fSCKPred} \rvert}{\text{\fSCKPred}} \times 100
  = \text{\fSCKErr}\,\%.
\end{equation}
