\section{SPI Waveform}
\label{sec:spi-waveform}

\paragraph{Group test byte.}
With $n_1 = \nOne$ and $n_2 = \nTwo$:
\begin{align}
  n_1 \oplus n_2 &= \texttt{\nOneBin} \oplus \texttt{\nTwoBin} = \texttt{\XorBin} = \XorDec, \\
  B &= \left[(n_1 \oplus n_2) + \text{0x3D}\right] \bmod 256
     = (\XorDec + 61) \bmod 256 = \text{\BHex}.
\end{align}
$B$ is neither 0x00 nor 0xFF, so $B = \text{\BHex} = \texttt{\BBin}_2$.

\scopefig{task3_sck_mosi.jpeg}{$B$ transmitted with CS high: CH1 = SCK (PB13),
  CH2 = MOSI (PB15).}{fig:task3}

\paragraph{SCK frequency.}
From \cref{fig:task3}, $T_\mathrm{SCK} = \TSCKMeas$~\si{\micro\second}, so
$f_\mathrm{SCK,measured} = 1/T_\mathrm{SCK} = \fSCKMeas$~kHz, and
\begin{equation}
  \%\,\mathrm{error}
  = \frac{\lvert f_\mathrm{measured} - f_\mathrm{predicted}\rvert}{f_\mathrm{predicted}} \times 100
  = \frac{\lvert \fSCKMeas - \fSCKPred \rvert}{\fSCKPred} \times 100
  = \fSCKErr\,\%.
\end{equation}
