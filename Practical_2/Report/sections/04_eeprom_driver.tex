\section{EEPROM Driver}
\label{sec:eeprom}

\paragraph{Group address.}
\begin{equation}
  A = (n_1 + 3n_2) \bmod 256 = (\nOne + 3 \times \nTwo) \bmod 256
    = \ADec = \text{\AHex}.
\end{equation}

\paragraph{Transaction sequence.}
Each transaction is framed by CS low \ldots\ CS high (\cref{tab:transactions}).
The fitted EEPROM uses a two-byte address, MSB first~\cite{readme}.

\begin{table}[htbp]
  \centering
  \caption{Write and read sequence, in order.}
  \label{tab:transactions}
  \small
  \begin{tabularx}{\linewidth}{@{}l l X r@{}}
    \toprule
    \textbf{Step} & \textbf{Opcode} & \textbf{MOSI $\rightarrow$ MISO} & \textbf{Clocks} \\
    \midrule
    1. WREN  & \fillin{0x..} & WREN & 8 \\
    2. WRITE & \fillin{0x..} & WRITE, A[15:8], A[7:0], $B$ & 32 \\
    3. RDSR (repeat until ready) & \fillin{0x..} & RDSR, dummy $\rightarrow$ status & 16 \\
    4. READ  & \fillin{0x..} & READ, A[15:8], A[7:0], dummy $\rightarrow$ data & 32 \\
    \bottomrule
  \end{tabularx}
\end{table}
